# The SDL3 + Dear ImGui front end (OPTICA_FRONTEND=SDL). Included from the top-level
# CMakeLists.txt, after the optica_core library. Builds:
#   Linux, Windows  the program "optica", with SDL linked in;
#   Android         libmain.so and libSDL3.so, which SDL's Java activity loads
#                   (the Gradle project in android/ runs this build);
#   iOS             the app "Optica.app", with SDL linked in, from the Xcode project
#                   CMake generates (see platform/ios/README.md).
#
# SDL3 and Dear ImGui are cloned from GitHub at configure time (git is needed), at fixed
# versions. On desktops an SDL3 already installed on the system (3.2 or newer) is used
# instead when CMake finds it. To build offline, point CMake at local copies:
#     -DFETCHCONTENT_SOURCE_DIR_SDL3=<SDL source>  -DFETCHCONTENT_SOURCE_DIR_IMGUI=<imgui source>

cmake_minimum_required(VERSION 3.22)   # the CMake Android Studio installs by default

include(FetchContent)

if(ANDROID)
    # SDL's Java side loads SDL as its own library, libSDL3.so
    set(SDL_SHARED ON)
    set(SDL_STATIC OFF)
else()
    # SDL linked into the program
    set(SDL_SHARED OFF)
    set(SDL_STATIC ON)
    find_package(SDL3 3.2 CONFIG QUIET)
endif()
set(SDL_TEST_LIBRARY OFF)

if(NOT TARGET SDL3::SDL3)
    FetchContent_Declare(SDL3
        GIT_REPOSITORY https://github.com/libsdl-org/SDL.git
        GIT_TAG        release-3.4.8        # keep android/app/src/main/java/org/libsdl at the same version
        GIT_SHALLOW    TRUE)
    FetchContent_MakeAvailable(SDL3)
endif()
FetchContent_Declare(imgui
    GIT_REPOSITORY https://github.com/ocornut/imgui.git
    GIT_TAG        v1.92.9b
    GIT_SHALLOW    TRUE)
FetchContent_MakeAvailable(imgui)

# ---- Dear ImGui (it has no CMake build of its own) --------------------------------
add_library(imgui STATIC
    ${imgui_SOURCE_DIR}/imgui.cpp
    ${imgui_SOURCE_DIR}/imgui_draw.cpp
    ${imgui_SOURCE_DIR}/imgui_tables.cpp
    ${imgui_SOURCE_DIR}/imgui_widgets.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_sdl3.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_sdlrenderer3.cpp
)
# SYSTEM: our warning options do not apply to Dear ImGui's headers
target_include_directories(imgui SYSTEM PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
target_link_libraries(imgui PUBLIC SDL3::SDL3)

# ---- the font, built into the program ------------------------------------------------
# Roboto Medium from Dear ImGui's misc/fonts, turned into a C++ array.
set(OPTICA_FONT ${imgui_SOURCE_DIR}/misc/fonts/Roboto-Medium.ttf)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${OPTICA_FONT})
file(READ ${OPTICA_FONT} fontHex HEX)
string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," fontBytes "${fontHex}")
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/embedded_font.cpp.new
    "// Generated by CMake from ${OPTICA_FONT}\n"
    "#include \"embedded_font.h\"\n"
    "extern const unsigned char EmbeddedFontData[] = {${fontBytes}};\n"
    "extern const std::size_t EmbeddedFontSize = sizeof EmbeddedFontData;\n")
configure_file(${CMAKE_CURRENT_BINARY_DIR}/embedded_font.cpp.new
               ${CMAKE_CURRENT_BINARY_DIR}/embedded_font.cpp COPYONLY)   # rewritten only when it changes

# ---- the program ------------------------------------------------------------------------
set(OPTICA_SDL_SOURCES
    main.cpp
    sdl_app.h
    sdl_app.cpp
    imgui_editor_view.h
    imgui_editor_view.cpp
    scene_painter.h
    scene_painter.cpp
    toolbar.h
    toolbar.cpp
    embedded_font.h
    ${CMAKE_CURRENT_BINARY_DIR}/embedded_font.cpp
)
if(ANDROID)
    add_library(optica SHARED ${OPTICA_SDL_SOURCES})
    set_target_properties(optica PROPERTIES OUTPUT_NAME main)    # libmain.so, as SDLActivity expects
elseif(IOS)
    # The app bundle: Info.plist, the icon (an asset catalog) and the signing settings
    # Xcode needs. Set OPTICA_IOS_TEAM to your Apple team ID, or pick the team in Xcode.
    set(OPTICA_BUNDLE_ID "com.example.optica" CACHE STRING "iOS bundle identifier (change it before publishing)")
    set(OPTICA_IOS_TEAM  ""                   CACHE STRING "Apple development team ID used to sign the app")
    set(OPTICA_IOS_ASSETS ${PROJECT_SOURCE_DIR}/platform/ios/Assets.xcassets)

    add_executable(optica MACOSX_BUNDLE ${OPTICA_SDL_SOURCES} ${OPTICA_IOS_ASSETS})
    set_source_files_properties(${OPTICA_IOS_ASSETS} PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
    set_target_properties(optica PROPERTIES
        OUTPUT_NAME                                       Optica
        MACOSX_BUNDLE_INFO_PLIST                          ${PROJECT_SOURCE_DIR}/platform/ios/Info.plist.in
        MACOSX_BUNDLE_GUI_IDENTIFIER                      ${OPTICA_BUNDLE_ID}
        MACOSX_BUNDLE_BUNDLE_NAME                         Optica
        MACOSX_BUNDLE_SHORT_VERSION_STRING                1.0
        MACOSX_BUNDLE_BUNDLE_VERSION                      1
        XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER         ${OPTICA_BUNDLE_ID}
        XCODE_ATTRIBUTE_TARGETED_DEVICE_FAMILY            "1,2"          # iPhone and iPad
        XCODE_ATTRIBUTE_ASSETCATALOG_COMPILER_APPICON_NAME AppIcon
        XCODE_ATTRIBUTE_CODE_SIGN_STYLE                   Automatic
        XCODE_ATTRIBUTE_DEVELOPMENT_TEAM                  "${OPTICA_IOS_TEAM}")
else()
    add_executable(optica ${OPTICA_SDL_SOURCES})
    set_target_properties(optica PROPERTIES WIN32_EXECUTABLE ON) # on Windows: no console window
endif()
target_include_directories(optica PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(optica PRIVATE optica_core imgui SDL3::SDL3)

if(MSVC)
    target_compile_options(optica PRIVATE /W3 /Zc:__cplusplus)
else()
    target_compile_options(optica PRIVATE -Wall)
endif()

# ---- tests: the view state, without SDL or Dear ImGui (not built for Android or iOS) -------
if(NOT ANDROID AND NOT IOS)
    add_executable(imgui_editor_view_test tests/imgui_editor_view_test.cpp imgui_editor_view.cpp)
    target_include_directories(imgui_editor_view_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(imgui_editor_view_test PRIVATE optica_core)
    add_test(NAME imgui_editor_view_test COMMAND imgui_editor_view_test)
endif()
